/* focus trap helper: returns a keydown handler to attach/detach */
function trapFocus(container){
  const focusables = container.querySelectorAll('a[href], button:not([disabled]), input, select, textarea, [tabindex]:not([tabindex="-1"])');
  if(!focusables.length) return null;
  const first = focusables[0], last = focusables[focusables.length-1];
  first.focus();
  function handler(e){
    if(e.key==='Tab'){
      if(e.shiftKey && document.activeElement===first){ e.preventDefault(); last.focus(); }
      else if(!e.shiftKey && document.activeElement===last){ e.preventDefault(); first.focus(); }
    }
  }
  container.addEventListener('keydown', handler);
  return handler;
}

function resetFilters(rerender=true){

  const si = document.getElementById('shop-search'); if(si) si.value='';
  const ii = document.getElementById('filter-instock'); if(ii) ii.checked=false;
  if(rerender) applyFilters();
}

function scoreProduct(p, ans){
  let score = 0, reasons = [];
  const priceNow = p.salePrice||p.price;
  const budgetRanges = { low:[0,30000], mid:[30000,60000], high:[60000,100000], lux:[100000,999999] };
  if(ans.budget && budgetRanges[ans.budget]){
    const [lo,hi] = budgetRanges[ans.budget];
    if(priceNow>=lo && priceNow<=hi){ score+=3; reasons.push('fits your budget'); }
    else if(Math.abs(priceNow-hi)<15000 || Math.abs(priceNow-lo)<15000){ score+=1; }
  }
  const styleMap = { classic:'heritage', sport:'pursuit', dress:'aurum', minimal:'noir' };
  if(ans.style && styleMap[ans.style]===p.collection){ score+=3; reasons.push(`matches your ${ans.style} style`); }
  const occasionMap = { daily:['heritage','noir'], work:['aurum','heritage'], active:['pursuit'], evening:['aurum'] };
  if(ans.occasion && occasionMap[ans.occasion]?.includes(p.collection)){ score+=2; reasons.push(`suits ${ans.occasion==='work'?'office wear':ans.occasion} `); }
  if(ans.movement && ans.movement!=='no-pref' && p.movement===ans.movement){ score+=2; reasons.push(`${p.movement.toLowerCase()} movement as requested`); }
  if(ans.material && ans.material!=='no-pref' && p.material===ans.material){ score+=2; reasons.push(`${p.material.toLowerCase()} case`); }
  if(ans.gender){ if(ans.gender==='Unisex' || p.gender===ans.gender || p.gender==='Unisex'){ score+=1; } }
  if(p.stock<=0) score -= 5;
  return { score, reasons };
}

async function placeOrder(e){
  e.preventDefault();
  if(!state.cart.length){ showToast('Your bag is empty.', 'err'); return false; }
  const method = document.querySelector('input[name=paymethod]:checked')?.value || 'cod';
  if((method==='jazzcash'||method==='easypaisa')){
    const sender = document.getElementById('pc-sender').value.trim();
    const txn = document.getElementById('pc-txn').value.trim();
    if(!sender || !txn){ showToast('Please complete the payment confirmation fields.', 'err'); return false; }
  }
  const methodLabel = {cod:'Cash on Delivery', jazzcash:'JazzCash', easypaisa:'Easypaisa', bank:'Bank Transfer'}[method] || method;
  const submitBtn = document.querySelector('#checkout-form button[type=submit]');
  if(submitBtn){ submitBtn.disabled = true; submitBtn.textContent = 'Placing Order…'; }

  const payload = {
    items: state.cart.map(i=>({ id:i.id, qty:i.qty })), // price is intentionally NOT sent — server recomputes it
    customer: {
      name: document.getElementById('co-name').value.trim(),
      phone: document.getElementById('co-phone').value.trim(),
      email: document.getElementById('co-email').value.trim(),
      address: document.getElementById('co-address').value.trim(),
      city: document.getElementById('co-city').value.trim(),
    },
    paymentMethod: method,
    paymentConfirmation: (method==='jazzcash'||method==='easypaisa') ? {
      senderName: document.getElementById('pc-sender').value.trim(),
      transactionId: document.getElementById('pc-txn').value.trim(),
    } : null,
  };

  const result = await apiPost('/orders', payload);
  let orderNumber, total;

  if(result.ok){
    orderNumber = result.data.order.orderNumber;
    total = result.data.order.total;
  } else if(result.offline){
    // Backend not running — demo fallback so the flow still completes end-to-end.
    orderNumber = document.getElementById('pc-order').value || ('CHR-' + Date.now().toString().slice(-8));
    total = cartTotal();
  } else {
    if(submitBtn){ submitBtn.disabled = false; submitBtn.textContent = 'Place Order'; }
    showToast(result.error, 'err');
    return false;
  }

  document.getElementById('confirmation-detail').textContent = result.ok
    ? `Order ${orderNumber} for ${fmtPrice(total)} was received and verified by the CHRONOVA backend via ${methodLabel}. Qasim will confirm by phone or WhatsApp shortly.`
    : `Order ${orderNumber} for ${fmtPrice(total)} was captured via ${methodLabel} in local demo mode (no backend connected). Connect the /backend service to store real orders.`;
  const waMsg = encodeURIComponent(`Hi CHRONOVA, I just placed order ${orderNumber} (${methodLabel}, ${fmtPrice(total)}). Please confirm.`);
  document.getElementById('confirmation-whatsapp').onclick = ()=>window.open(`https://wa.me/923456607934?text=${waMsg}`, '_blank');
  state.cart = [];
  saveLS(LS_KEYS.cart, state.cart);
  updateBadges();
  navigate('confirmation');
  return false;
}

async function submitContactForm(e){
  e.preventDefault();
  const name = document.getElementById('cf-name').value.trim();
  const msg = document.getElementById('cf-message').value.trim();
  if(!name || !msg){ showToast('Please fill in your name and message.', 'err'); return false; }
  const payload = {
    name, msg,
    phone: document.getElementById('cf-phone').value.trim(),
    email: document.getElementById('cf-email').value.trim(),
    subject: document.getElementById('cf-subject').value,
    message: msg,
  };
  const result = await apiPost('/contact', payload);
  if(result.ok || result.offline){
    showToast('Message received — Qasim typically replies within one business day.');
    e.target.reset();
  } else {
    showToast(result.error, 'err');
  }
  return false;
}
async function subscribeNewsletter(e, source){
  e.preventDefault();
  const input = document.getElementById('newsletter-email-'+source);
  const email = input.value.trim();
  if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)){ showToast('Please enter a valid email address.', 'err'); return false; }
  const result = await apiPost('/newsletter', { email });
  if(result.ok || result.offline){
    const list = loadLS(LS_KEYS.newsletter, []);
    if(!list.includes(email)) list.push(email);
    saveLS(LS_KEYS.newsletter, list);
    showToast('Subscribed — welcome to the CHRONOVA list.');
    input.value='';
  } else {
    showToast(result.error, 'err');
  }
  return false;
}
